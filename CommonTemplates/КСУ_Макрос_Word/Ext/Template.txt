Sub AddLinkToInstruction()
    Dim form1 As New Instructions_ksu
    Dim LinkArray(1000)
    Dim ReadData As String
    
    file = ActiveDocument.Path + "\inst_list.txt"

    LineCount = 0
    Open file For Input As #1
    Do Until EOF(1)
        Line Input #1, ReadData
        If Not ReadData = "" Then
            tmp = Split(ReadData, "|^|")
            LinkArray(LineCount) = tmp(1)
            form1.ListBox1.AddItem tmp(0), LineCount
            LineCount = LineCount + 1
        End If
    Loop
    Close #1
    
    form1.Show
    
    If form1.Res = 1 And form1.ListBox1.ListIndex > -1 Then
        inst_name = form1.ListBox1.Value
        inst_url = LinkArray(form1.ListBox1.ListIndex)
        
        ActiveDocument.Hyperlinks.Add Anchor:=Selection.Range, Address:=inst_url, _
            SubAddress:="", ScreenTip:="", TextToDisplay:=inst_name
    End If
End Sub
