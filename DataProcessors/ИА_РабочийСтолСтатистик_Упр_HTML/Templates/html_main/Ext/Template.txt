function GetTrendCoord(data, plotObj) {
	if (data == undefined){
		var axX = plotObj.getXAxes()[0];
		var optsX = axX.options;
		var range_min = optsX.min;
		var range_max = optsX.max;
		var data = [];
		
		fullData = plotObj.getData()[0];
		for(var m = 0; m < fullData.data.length; m++){
			if(fullData.data[m][0] != null && fullData.data[m][1] != null){
				if (range_min == null || fullData.data[m][0] >= range_min && fullData.data[m][0] <= range_max){					  
					data.push(fullData.data[m]);									  
				}
			
			}
		}
	};
	
	var ii=0, x, y, x0, x1, y0, y1, dx,
		m = 0, b = 0, cs, ns,
		n = data.length, Sx = 0, Sy = 0, Sxy = 0, Sx2 = 0, S2x = 0;
		
	// Not enough data or disabled
	if(n < 2 ) return;
	
	// Do math stuff
	for(ii;ii<data.length;ii++){
		x = data[ii][0];
		y = data[ii][1];
		Sx += x;
		Sy += y;
		Sxy += (x*y);
		Sx2 += (x*x);
	}
	// Calculate slope and intercept
	m = (n*Sx2 - S2x) != 0 ? (n*Sxy - Sx*Sy) / (n*Sx2 - Sx*Sx) : 0;
	b = (Sy - m*Sx) / n;
	
	// Calculate minimal coordinates to draw the trendline
	dx = parseFloat(data[1][0]) - parseFloat(data[0][0]);
	
	//x0 = parseFloat(data[0][0])-dx;
	x0 = parseFloat(data[0][0]);
	
	y0 = parseFloat(m*x0 + b);
	
	//x1 = parseFloat(data[ii-1][0])+dx;
	x1 = parseFloat(data[ii-1][0]);
	
	y1 = parseFloat(m*x1 + b);

	return [[x0,y0],[x1,y1]];
};

function zoom(ranges, plotN) {
	// set X range
	var XAxes = plotN.getXAxes();
	var axis;
	var opts;
	
	for(i = 0; i < XAxes.length; i++){
	    axis = XAxes[i];
	    opts = axis.options;
	    
	    xaxisnum = (i == 0) ? "" : i+1;
	    xaxisname = "x"+xaxisnum+"axis";
	    opts.min = ranges[xaxisname].from;
	    opts.max = ranges[xaxisname].to;
	}
	// set Y range
	SetYzoom(plotN, ranges);
	Refresh(plotN);
};

function SetReverse(opts, reversed) {				
	if (reversed){
		opts.yaxis.transform = function (v) { return -v; };
		opts.yaxis.inverseTransform = function (v) { return -v; };
	}
	else{
		opts.yaxis.transform = undefined;
		opts.yaxis.inverseTransform = undefined;
	};
};

function zoomOut(plotN) {

	if (points.length > 0)

	  }
		// set Y range
		SetYzoom(plotN);
		Refresh(plotN);
	};
};

function SetYzoom(plotN, ranges) {
	var minV = 100000000000;
	var maxV = -100000000000;
	var points = plotN.getData();
	var options = plotN.getOptions();
	var addScale = options.addScale;
	
    for(var k = 0; k < points.length; k++){
	   if ('trendLine' in points[k] && points[k].trendLine) {
		 continue;
	   };
	   for(var m = 0; m < points[k].data.length; m++){
		 if(points[k].data[m] != null && points[k].data[m][0] != null && points[k].data[m][1] != null){				  
			var curX = points[k].data[m][0];
			if (ranges === undefined || (curX >= ranges.xaxis.from && curX <= ranges.xaxis.to)){
			  var curVal = points[k].data[m][1];								  
			   if (curVal >= maxV){
						maxV = curVal;
					}
					if (curVal <= minV){
						minV = curVal;
			  }
			}
		  }
		}   
	 }
	
	if(mainScale)
	{
	    if(!(options.yaxis.min == 0)) 
		{
	        minV = options.yaxis.min;
	    }
	    
	    if(!(options.yaxis.max == 0)) 
		{
	        maxV = options.yaxis.max;
	    }
	}
	else
	{
	  if(addScale !== 0){
	     minV_ = minV -(addScale/100)*maxV;
	     if((minV >=0) && (minV_ < 0)){
	        minV = 0;
	     }
	     else{
	        minV = minV_;   
	     }
	     maxV = ((100 + addScale)/100)*maxV;
	  }
	}
	
	var axisY = plotN.getYAxes()[0];
	var optsY = axisY.options;
	optsY.min = minV;
	optsY.max = maxV;
};
